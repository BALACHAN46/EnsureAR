<script>
    // Variable to track the current Material Mode ('Diamond' or 'Gold')
    var currentMaterialMode = 'Diamond'; // Default

    // Function to handle clicks from Swiper side menu (Icons)
    function handleCategoryClick(baseCategory) {
        // Construct the full ID based on mode + base category
        let finalCategory = baseCategory;

        if (currentMaterialMode === 'Gold') {
            // Map the generic base name to the Gold specific key
            switch (baseCategory) {
                case 'Necklaces': finalCategory = 'Gold_Necklaces'; break;
                case 'Chains':    finalCategory = 'Gold_Chains'; break; // New Case
                case 'Earring':   finalCategory = 'Gold_Earring'; break;
                case 'NosePin':   finalCategory = 'Gold_NosePin'; break;
                case 'Sets':      finalCategory = 'Gold_Sets'; break;
                case 'Bracelets': finalCategory = 'Gold_Bracelets'; break;
                case 'Rings':     finalCategory = 'Gold_Rings'; break;
            }
        }
        
        showHideImages(finalCategory);
    }

    async function showHideImages(category) {
        // Defines items that require WRIST camera logic
        const wristHandItems = ['Watches', 'Bangles', 'Rings', 'Bracelets', 'Gold_Bracelets', 'Gold_Rings'];
        
        // Define all carousel IDs including the new Chains and Gold Chains
        var productCarousels = $('#div-necklace, #divChains, #divEaring, #divNosePin, #divSets, #divBracelets, #divRings, #divEyewear, #divWatches, #divBangles, #div-gold-necklace, #div-gold-chains, #div-gold-earring, #div-gold-nosepin, #div-gold-sets, #div-gold-bracelets, #div-gold-rings');
        var canvasElement = document.getElementById('canvas');

        // Hide all carousels first
        productCarousels.css('display', 'none'); 

        // Select the specific carousel based on Category string
        var selectedCarousel;
        
        // --- DIAMOND MAPPINGS ---
        if (category === 'Necklaces') selectedCarousel = $('#div-necklace');
        else if (category === 'Chains') selectedCarousel = $('#divChains'); // New
        else if (category === 'Earring') selectedCarousel = $('#divEaring');
        else if (category === 'NosePin') selectedCarousel = $('#divNosePin');
        else if (category === 'Sets') selectedCarousel = $('#divSets');
        else if (category === 'Bracelets') selectedCarousel = $('#divBracelets');
        else if (category === 'Rings') selectedCarousel = $('#divRings');
        
        // --- GOLD MAPPINGS ---
        else if (category === 'Gold_Necklaces') selectedCarousel = $('#div-gold-necklace');
        else if (category === 'Gold_Chains') selectedCarousel = $('#div-gold-chains'); // New
        else if (category === 'Gold_Earring') selectedCarousel = $('#div-gold-earring');
        else if (category === 'Gold_NosePin') selectedCarousel = $('#div-gold-nosepin');
        else if (category === 'Gold_Sets') selectedCarousel = $('#div-gold-sets');
        else if (category === 'Gold_Bracelets') selectedCarousel = $('#div-gold-bracelets');
        else if (category === 'Gold_Rings') selectedCarousel = $('#div-gold-rings');
        
        // --- COMMON ---
        else if (category === 'Eyewear') selectedCarousel = $('#divEyewear');
        else if (category === 'Watches') selectedCarousel = $('#divWatches');
        else if (category === 'Bangles') selectedCarousel = $('#divBangles');

        // Display and Init Carousel
        if (selectedCarousel && selectedCarousel.length > 0) {
            selectedCarousel.css('display', 'block'); 
            if (selectedCarousel.hasClass('owl-loaded')) {
                await Sleepms(50);
                selectedCarousel.trigger('refresh.owl.carousel');
            } else {
                selectedCarousel.owlCarousel({
                    items: 3, loop: false, autoWidth: false, nav: false, dots: false, margin: 10, autoplay: false,
                    responsiveClass: true, responsive: { 0: { items: 3, nav: false }, 400: { items: 4, nav: false }, 600: { items: 5, nav: false }, 992: { items: 3, nav: false } },
                });
            }
        } 

        // Camera positioning Logic (Face vs Wrist)
        if (canvasElement) {
            const container = document.getElementById('camera-container');
            if (wristHandItems.includes(category)) {
                canvasElement.classList.remove('camera-default');
                canvasElement.classList.add('camera-wrist');
                container?.classList.remove('camera-default');
                container?.classList.add('camera-wrist');
            } else {
                canvasElement.classList.remove('camera-wrist');
                canvasElement.classList.add('camera-default');
                container?.classList.remove('camera-wrist');
                container?.classList.add('camera-default');
            }
            setTimeout(() => {
                if (window.cameraKitSession?.resize) {
                    window.cameraKitSession.resize();
                }
            }, 100);
        }

        $('#modalCenter').modal('show'); 
    }

    function Sleepms(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }

    $(document).ready(function () {
        $('#canvas').addClass('camera-default');

        const queryString = window.location.search;
        const urlParams = new URLSearchParams(queryString);
        const product = urlParams.get('Item'); 

        // Determine Mode based on URL Parameter
        if (product && product.includes('Gold_')) {
            currentMaterialMode = 'Gold';
        } else {
            currentMaterialMode = 'Diamond';
        }

        console.log("Current Mode: " + currentMaterialMode);

        if(product) {
            showHideImages(product);
        } else {
            // Default if nothing passed
            showHideImages('Necklaces');
        }
    });
